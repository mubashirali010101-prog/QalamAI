// 1. Configuration (Apni real key yahan paste karein) const GEMINI_API_KEY = "YOUR_REAL_API_KEY_HERE"; const GEMINI_ENDPOINT = `https://googleapis.com{GEMINI_API_KEY}`; // 2. Selectors (Aapke UI design ke mutabiq targets) const generateBtn = document.querySelector('button'); // Aapka "Generate Magic" button const promptInput = document.querySelector('input[type="text"]') || document.querySelector('textarea'); // Input field const outputContainer = document.querySelector('.border-dashed') || document.querySelector('#output'); // QalamAI Engine Output area // 3. Main Logic Execution if (generateBtn && promptInput && outputContainer) { generateBtn.addEventListener('click', async () => { const promptText = promptInput.value.trim(); if (!promptText) { alert("Please enter a prompt first!"); return; } // Loading State: Button ko freeze karein taake user baar baar click na kare generateBtn.disabled = true; const originalBtnText = generateBtn.innerHTML; generateBtn.innerHTML = "Generating Magic... ✨"; outputContainer.innerHTML = `QalamAI is thinking and crafting your copy...`; try { // API Call Execution const response = await fetch(GEMINI_ENDPOINT, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ contents: [{ parts: [{ text: promptText }] }] }) }); if (!response.ok) { throw new Error(`HTTP Error Status: ${response.status}`); } const data = await response.json(); // ✅ EXACT FIXED GEMINI PARSING FORMAT (No more hanging) if (data.candidates && data.candidates[0].content.parts[0].text) { const aiResult = data.candidates[0].content.parts[0].text; outputContainer.innerText = aiResult; } else { throw new Error("Invalid response structure from Gemini API"); } } catch (error) { console.error("QalamAI Suite Error:", error); outputContainer.innerHTML = `Failed to generate content. Error: ${error.message}`; } finally { // Reset State: UI ko wapas active karein generateBtn.disabled = false; generateBtn.innerHTML = originalBtnText; } }); } else { console.error("QalamAI Setup Error: HTML Input, Button, or Output elements not found. Please verify DOM classes/tags."); }
